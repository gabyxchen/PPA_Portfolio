# Clean
evict_raw <- evict_raw %>%
  mutate(
    month = as.character(month),
    month_date = mdy(paste0("01/", month)),   # turn "01/2020" → "01/01/2020"
    month_num = month(month_date),            # 1–12
    year = year(month_date),                  # also useful
    filings = as.numeric(filings_2020)
  )

# Remove sealed
evict_cleaned <- evict_raw %>%
  filter(GEOID != "sealed")

# Keep a copy of monthly data (important!)
evict_monthly <- evict_cleaned

# Aggregate to monthly
evict_m <- evict_cleaned %>%
  group_by(GEOID, month_date) %>%
  summarise(
    filings = sum(filings, na.rm = TRUE),
    .groups = "drop"
  )

# Complete dataset
evict_m <- evict_m %>%
  left_join(
    evict_cleaned %>%
      dplyr::select(GEOID, month_date, type, racial_majority, month_num, year) %>%
      distinct(),
    by = c("GEOID", "month_date")
  )

# Filter time
evict <- evict_m %>%
  filter(
    month_date >= as.Date("2023-03-01") &
    month_date <= as.Date("2026-03-31")
  )

# Summary
summary_table <- tibble(
  step = c(
    "Original data",
    "After removing 'sealed'",
    "After time filter (2023-03 to 2026-03)"
  ),
  n_rows = c(
    nrow(evict_raw),
    nrow(evict_cleaned),
    nrow(evict)
  )
)
