renters <- renters %>%
  dplyr::select(GEOID, renters = estimate)

# Join to dataset
evict <- evict %>%
  left_join(renters, by = "GEOID")

# Normaliza filings
evict <- evict %>%
  mutate(
    filings_per_1k_renters = (filings_2020 / renters) * 1000
  )

# Compute tract-level average rate
evict_clean_rate <- evict %>%
  filter(!is.na(renters) & renters > 0)

evict_clean_rate <- evict_clean_rate %>%
  filter(renters > 100)   # adjust threshold

top10_rate <- evict_clean_rate %>%
  group_by(GEOID) %>%
  summarise(
    avg_rate = mean(filings_per_1k_renters, na.rm = TRUE),
    total_filings = sum(filings_2020, na.rm = TRUE),
    renters = first(renters),
    .groups = "drop"
  ) %>%
  arrange(desc(avg_rate)) %>%
  slice(1:10)

Dynamics

  • Create multiple lag variables (1 month lag, 1 quarter lag)
  • Visualize Changes
Code 
evict <- evict %>%
  arrange(GEOID, month_date) %>%
  group_by(GEOID)

# 1 month lag
evict <- evict %>%
  mutate(
    lag_1m = lag(filings_2020, 1)
  )

# Quarter lag
evict <- evict %>%
  mutate(
    lag_3m = lag(filings_2020, 3)
  )

# Create change variables
evict <- evict %>%
  mutate(
    change_1m = filings_2020 - lag_1m,
    change_3m = filings_2020 - lag_3m
  )

# Visualize 1 month change
ggplot(evict, aes(x = change_1m)) +
  geom_histogram(bins = 50) +
  labs(title = "Monthly Change in Filings")
